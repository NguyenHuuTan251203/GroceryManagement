
INSERT INTO HoaDon (SoHD, NgayBan, MaKH, MaNV, ChiTietHoaDon)
SELECT 
    @SoHD, 
    @NgayBan, 
    KH.MaKH, 
    NV.MaNV, 
    CONCAT(HH.Mahang, ',', @Soluong, ',', @DonGiaBan)
FROM 
    KhachHang KH
JOIN 
    NhanVien NV ON NV.MaNV = @MaNV
JOIN 
    HangHoa HH ON HH.TenKH = KH.TenKH  
WHERE 
    KH.MaKH = @MaKH;


use QuanLyTapHoa
go
CREATE TRIGGER trg_InsertCT_HoaDon
ON HoaDon
AFTER INSERT
AS
BEGIN
    -- Khai báo các biến
    DECLARE @SoHD INT;
    DECLARE @ChiTietHoaDon NVARCHAR(MAX);
    
    -- Lấy giá trị từ bảng inserted
    SELECT @SoHD = SoHD, @ChiTietHoaDon = ChiTietHoaDon FROM inserted;

    -- Cắt chuỗi trong cột ChiTietHoaDon thành các phần tử
    -- Giả sử chuỗi có cấu trúc "Mahang,Soluong,DonGiaBan"
    DECLARE @Mahang NVARCHAR(50), @Soluong INT, @DonGiaBan DECIMAL(18, 2);

    -- Tách chuỗi và gán vào các biến
    SET @Mahang = PARSENAME(REPLACE(@ChiTietHoaDon, ',', '.'), 3);  -- Lấy Mahang
    SET @Soluong = CAST(PARSENAME(REPLACE(@ChiTietHoaDon, ',', '.'), 2) AS INT);  -- Lấy Soluong
    SET @DonGiaBan = CAST(PARSENAME(REPLACE(@ChiTietHoaDon, ',', '.'), 1) AS DECIMAL(18, 2));  -- Lấy DonGiaBan

    -- Chèn vào bảng CT_HoaDon
    INSERT INTO CT_HoaDon (SoHD, Mahang, Soluong, DonGiaBan)
    VALUES (@SoHD, @Mahang, @Soluong, @DonGiaBan);
END;
